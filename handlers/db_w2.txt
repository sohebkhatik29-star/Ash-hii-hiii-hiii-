
    # -------------------- search --------------------
    async def search_movies(self, query: str, page: int = 1, per_page: int = 10) -> dict:
        q = (query or "").strip()
        if not q:
            return {"results": [], "total": 0, "page": page, "per_page": per_page}
        tokens = [
            t.lower()
            for t in re.split(r"[\s\-_\.]+", q)
            if t and t.lower() not in STOPWORDS and len(t) > 1
        ]
        if not tokens:
            tokens = [q.lower()]
        results = []
        total = 0
        if self.has_mongo and self.movies_col is not None:
            try:
                and_clauses = []
                for tok in tokens:
                    and_clauses.append({
                        "$or": [
                            {"title": {"$regex": re.escape(tok), "$options": "i"}},
                            {"normalized_title": {"$regex": re.escape(tok), "$options": "i"}},
                            {"file_name": {"$regex": re.escape(tok), "$options": "i"}},
                            {"caption": {"$regex": re.escape(tok), "$options": "i"}},
                        ]
                    })
                filt = {"$and": and_clauses} if len(and_clauses) > 1 else (and_clauses[0] if and_clauses else {})
                total = await self.movies_col.count_documents(filt)
                cursor = (
                    self.movies_col.find(filt)
                    .sort("added_at", -1)
                    .skip(max(0, (page - 1) * per_page))
                    .limit(per_page)
                )
                async for doc in cursor:
                    doc["_id"] = str(doc.get("_id", ""))
                    results.append(doc)
            except Exception as e:
                logger.error("search_movies: %s", e)
        else:
            for m in self._mem_movies:
                hay = f"{m.get('title','')} {m.get('normalized_title','')} {m.get('file_name','')} {m.get('caption','')}".lower()
                if all(t in hay for t in tokens):
                    results.append(m)
            total = len(results)
            results = results[(page - 1) * per_page: page * per_page]
        return {"results": results, "total": total, "page": page, "per_page": per_page}

    # -------------------- users / bans --------------------
    async def add_user(self, user_id: int, name: str = ""):
        if self.has_mongo and self.users_col is not None:
            try:
                await self.users_col.update_one(
                    {"user_id": int(user_id)},
                    {
                        "$set": {"user_id": int(user_id), "name": name, "last_seen": int(time.time())},
                        "$setOnInsert": {"joined_at": int(time.time()), "banned": False},
                    },
                    upsert=True,
                )
            except Exception as e:
                logger.warning("add_user: %s", e)

    async def is_user_exist(self, user_id: int) -> bool:
        if self.has_mongo and self.users_col is not None:
            try:
                doc = await self.users_col.find_one({"user_id": int(user_id)})
                return bool(doc)
            except Exception:
                pass
        return False

    async def get_ban_status(self, user_id: int) -> dict:
        if user_id in getattr(Config, "BANNED_USERS", set()):
            return {"is_banned": True, "reason": "config"}
        if self.has_mongo and self.users_col is not None:
            try:
                doc = await self.users_col.find_one({"user_id": int(user_id)})
                if doc and doc.get("banned"):
                    return {"is_banned": True, "reason": doc.get("ban_reason", "")}
            except Exception:
                pass
        return {"is_banned": False, "reason": ""}

    async def is_banned(self, user_id: int) -> bool:
        st = await self.get_ban_status(user_id)
        return bool(st.get("is_banned"))

    async def ban_user(self, user_id: int, reason: str = ""):
        if self.has_mongo and self.users_col is not None:
            try:
                await self.users_col.update_one(
                    {"user_id": int(user_id)},
                    {"$set": {"banned": True, "ban_reason": reason, "banned_at": int(time.time())}},
                    upsert=True,
                )
            except Exception as e:
                logger.warning("ban_user: %s", e)

    async def unban_user(self, user_id: int):
        if self.has_mongo and self.users_col is not None:
            try:
                await self.users_col.update_one(
                    {"user_id": int(user_id)},
                    {"$set": {"banned": False}, "$unset": {"ban_reason": "", "banned_at": ""}},
                )
            except Exception as e:
                logger.warning("unban_user: %s", e)

    async def add_warn(self, user_id: int, by: int = 0, reason: str = "") -> int:
        return await self.add_warning(user_id, by, reason)

    async def add_warning(self, user_id: int, by: int = 0, reason: str = "") -> int:
        count = 0
        if self.has_mongo and self.warns_col is not None:
            try:
                await self.warns_col.insert_one({
                    "user_id": int(user_id), "by": by, "reason": reason, "at": int(time.time())

            except Exception as e:
                logger.warning("add_warning: %s", e)

    # -------------------- active sessions (session_mgr) --------------------
    async def set_active_session(self, admin_id: int, data: dict):
        data = dict(data or {})
        data["admin_id"] = int(admin_id)
        data["updated_at"] = int(time.time())
        self._mem_sessions[int(admin_id)] = data
        if self.has_mongo and self.sessions_col is not None:
            try:
                await self.sessions_col.update_one(
                    {"admin_id": int(admin_id)},
                    {"$set": data},
                    upsert=True,
                )
            except Exception as e:
                logger.warning("set_active_session: %s", e)

    async def get_active_session(self, admin_id: int) -> Optional[dict]:
        if int(admin_id) in self._mem_sessions:
            return self._mem_sessions[int(admin_id)]
        if self.has_mongo and self.sessions_col is not None:
            try:
                doc = await self.sessions_col.find_one({"admin_id": int(admin_id)})
                if doc:
                    self._mem_sessions[int(admin_id)] = doc
                    return doc
            except Exception:
                pass
        return None

    async def clear_active_session(self, admin_id: int):
        self._mem_sessions.pop(int(admin_id), None)
        if self.has_mongo and self.sessions_col is not None:
            try:
                await self.sessions_col.delete_one({"admin_id": int(admin_id)})
            except Exception:
                pass

    async def get_all_active_sessions(self) -> List[dict]:
        out = list(self._mem_sessions.values())
        if self.has_mongo and self.sessions_col is not None:
            try:
                async for d in self.sessions_col.find({}):
                    aid = int(d.get("admin_id", 0))
                    if aid and aid not in self._mem_sessions:
                        out.append(d)
                        self._mem_sessions[aid] = d
            except Exception:
                pass
        return out

    async def save_session(self, admin_id: int, data: dict):
        await self.set_active_session(admin_id, data)

    async def get_session(self, admin_id: int):
        return await self.get_active_session(admin_id)

    async def delete_session(self, admin_id: int):
        await self.clear_active_session(admin_id)

    async def list_sessions(self) -> List[dict]:
        return await self.get_all_active_sessions()
